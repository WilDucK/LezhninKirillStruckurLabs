## Обоснование выбора структуры данных

### Был выбран **односвязный список (LinkedList)**, так как в плейлисте важны операции вставки и удаления треков. 
### Связный список позволяет удалить элемент за O(1) после нахождения позиции. 
### Перемешивание (shuffle) реализовано через вспомогательный массив, что обеспечивает эффективность O(n). 
### Связный список более эффективен при частых вставках/удалениях.

### Класс Track - представление музыкального трека

```python
class Track:
    def __init__(self, name: str, duration: int, genre: str, rating: float):
        self.name = name
        self.duration = duration
        self.genre = genre
        self.rating = rating
        self.next = None

    def __str__(self):
        minutes = self.duration // 60
        seconds = self.duration % 60
        return f"{self.name} | {minutes}:{seconds:02d} | {self.genre} | {self.rating}"

    def to_base64_string(self) -> str:
        data = f"{self.name}|{self.duration}|{self.genre}|{self.rating}"
        return base64.b64encode(data.encode()).decode()

    @classmethod
    def from_base64_string(cls, base64_str: str) -> 'Track':
        data = base64.b64decode(base64_str).decode()
        name, duration, genre, rating = data.rsplit('|', 3)
        return cls(name, int(duration), genre, float(rating))
```

### Класс LinkedList - реализация односвязного списка

```python
class LinkedList:
    def __init__(self):
        self.head = None
        self._size = 0

    def __len__(self):
        return self._size

    def __iter__(self):
        current = self.head
        while current:
            yield current.track
            current = current.next

    def append(self, track: Track) -> None:
        new_node = Node(track)
        if not self.head:
            self.head = new_node
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_node
        self._size += 1

    def append_to_beginning(self, track: Track) -> None:
        new_node = Node(track)
        new_node.next = self.head
        self.head = new_node
        self._size += 1

    def remove_by_name(self, name: str) -> bool:
        if not self.head:
            return False
        if self.head.track.name == name:
            self.head = self.head.next
            self._size -= 1
            return True
        current = self.head
        while current.next:
            if current.next.track.name == name:
                current.next = current.next.next
                self._size -= 1
                return True
            current = current.next
        return False

    def remove_by_index(self, index: int) -> bool:
        if index < 0 or index >= self._size:
            return False
        if index == 0:
            self.head = self.head.next
            self._size -= 1
            return True
        current = self.head
        for i in range(index - 1):
            current = current.next
        current.next = current.next.next
        self._size -= 1
        return True

    def get_by_index(self, index: int) -> Optional[Track]:
        if index < 0 or index >= self._size:
            return None
        current = self.head
        for i in range(index):
            current = current.next
        return current.track

    def to_list(self) -> list:
        return [track for track in self]

    def shuffle(self) -> None:
        if self._size <= 1:
            return
        tracks_list = self.to_list()
        random.shuffle(tracks_list)
        self.head = None
        self._size = 0
        for track in tracks_list:
            self.append(track)

    def clear(self) -> None:
        self.head = None
        self._size = 0
```
### Класс Playlist - управление плейлистом

```python
class Playlist:
    def __init__(self):
        self.tracks = LinkedList()
        self.repeat_mode = "none"

    def add_track(self, name: str, duration: int, genre: str, rating: float, to_beginning: bool = False) -> None:
        track = Track(name, duration, genre, rating)
        if to_beginning:
            self.tracks.append_to_beginning(track)
            print(f"[+] Добавлен трек в начало: {track}")
        else:
            self.tracks.append(track)
            print(f"[+] Добавлен трек: {track}")

    def remove_track(self, name: str) -> None:
        if self.tracks.remove_by_name(name):
            print(f"[-] Удален трек: {name}")
        else:
            print(f"[!] Трек '{name}' не найден")

    def remove_track_by_index(self, index: int) -> None:
        if self.tracks.remove_by_index(index):
            print(f"[-] Удален трек с индексом {index}")
        else:
            print(f"[!] Неверный индекс")

    def shuffle(self) -> None:
        print("[*] Перемешивание плейлиста...")
        self.tracks.shuffle()
        print("[+] Плейлист перемешан")

    def set_repeat_mode(self, mode: str) -> None:
        if mode in ["none", "one", "all"]:
            self.repeat_mode = mode
            mode_names = {"none": "без повтора", "one": "повтор одного трека", "all": "повтор всего плейлиста"}
            print(f"[*] Режим повтора установлен: {mode_names[mode]}")
        else:
            print(f"[!] Неверный режим повтора. Используйте: none, one, all")

    def get_repeat_mode_display(self) -> str:
        modes = {"none": "[ ] Без повтора", "one": "[1] Повтор одного", "all": "[A] Повтор всех"}
        return modes.get(self.repeat_mode, "[ ] Без повтора")

    def get_tracks_by_rating(self, min_rating: float) -> list:
        filtered = [track for track in self.tracks if track.rating >= min_rating]
        if filtered:
            print(f"\n[+] Треки с рейтингом не ниже {min_rating}:")
            for i, track in enumerate(filtered, 1):
                print(f"   {i}. {track.name} - {track.rating}")
        else:
            print(f"[!] Нет треков с рейтингом не ниже {min_rating}")
        return filtered

    def print_playlist(self) -> None:
        if len(self.tracks) == 0:
            print("\n[ ] Плейлист пуст")
            return
        tracks_list = self.tracks.to_list()
        max_name_len = max(len(track.name) for track in tracks_list)
        max_duration_len = max(len(self._format_duration(track.duration)) for track in tracks_list)
        max_genre_len = max(len(track.genre) for track in tracks_list)
        max_rating_len = max(len(str(track.rating)) for track in tracks_list)
        max_name_len = max(max_name_len, len("НАЗВАНИЕ"))
        max_duration_len = max(max_duration_len, len("ДЛИТЕЛЬНОСТЬ"))
        max_genre_len = max(max_genre_len, len("ЖАНР"))
        max_rating_len = max(max_rating_len, len("РЕЙТИНГ"))
        total_width = max_name_len + max_duration_len + max_genre_len + max_rating_len + 13
        print("\n" + "=" * total_width)
        print(" МОЙ ПЛЕЙЛИСТ ".center(total_width))
        print("=" * total_width)
        header = f"| N | {'НАЗВАНИЕ'.ljust(max_name_len)} | {'ДЛИТЕЛЬНОСТЬ'.ljust(max_duration_len)} | {'ЖАНР'.ljust(max_genre_len)} | {'РЕЙТИНГ'.ljust(max_rating_len)} |"
        print(header)
        print("|---+-" + "-" * (max_name_len + 2) + "+-" + "-" * (max_duration_len + 2) + "+-" + "-" * (max_genre_len + 2) + "+-" + "-" * (max_rating_len + 2) + "-|")
        for i, track in enumerate(tracks_list, 1):
            duration_str = self._format_duration(track.duration)
            row = f"| {i:2} | {track.name.ljust(max_name_len)} | {duration_str.ljust(max_duration_len)} | {track.genre.ljust(max_genre_len)} | {str(track.rating).ljust(max_rating_len)} |"
            print(row)
        print("=" * total_width)
        print(f"\n Всего треков: {len(tracks_list)}")
        print(f" Режим повтора: {self.get_repeat_mode_display()}")

    @staticmethod
    def _format_duration(seconds: int) -> str:
        minutes = seconds // 60
        secs = seconds % 60
        return f"{minutes}:{secs:02d}"

    def save_to_file(self, filename: str) -> None:
        if len(self.tracks) == 0:
            print("[!] Плейлист пуст, нечего сохранять")
            return
        try:
            with open(filename, 'w', encoding='utf-8') as f:
                for track in self.tracks:
                    f.write(track.to_base64_string() + '\n')
            print(f"[+] Плейлист сохранен в файл: {filename}")
            print(f"    Сохранено треков: {len(self.tracks)}")
        except Exception as e:
            print(f"[!] Ошибка при сохранении: {e}")

    def load_from_file(self, filename: str) -> None:
        try:
            with open(filename, 'r', encoding='utf-8') as f:
                lines = f.readlines()
            self.tracks.clear()
            loaded_count = 0
            for line in lines:
                line = line.strip()
                if line:
                    try:
                        track = Track.from_base64_string(line)
                        self.tracks.append(track)
                        loaded_count += 1
                    except Exception as e:
                        print(f"[!] Ошибка при загрузке трека: {e}")
            print(f"[+] Плейлист загружен из файла: {filename}")
            print(f"    Загружено треков: {loaded_count}")
        except FileNotFoundError:
            print(f"[!] Файл '{filename}' не найден")
        except Exception as e:
            print(f"[!] Ошибка при загрузке: {e}")

    def get_statistics(self) -> dict:
        if len(self.tracks) == 0:
            return {
                'total_tracks': 0,
                'total_duration': 0,
                'total_duration_formatted': "0:00",
                'average_rating': 0.0,
                'genres': {},
                'max_duration_track': None,
                'max_rating_track': None
            }
        total_duration = 0
        total_rating = 0
        genres = {}
        max_duration_track = None
        max_rating_track = None
        max_duration = 0
        max_rating = -1.0
        for track in self.tracks:
            total_duration += track.duration
            total_rating += track.rating
            genres[track.genre] = genres.get(track.genre, 0) + 1
            if track.duration > max_duration:
                max_duration = track.duration
                max_duration_track = track
            if track.rating > max_rating:
                max_rating = track.rating
                max_rating_track = track
        minutes = total_duration // 60
        seconds = total_duration % 60
        return {
            'total_tracks': len(self.tracks),
            'total_duration': total_duration,
            'total_duration_formatted': f"{minutes}:{seconds:02d}",
            'average_rating': round(total_rating / len(self.tracks), 2),
            'genres': genres,
            'max_duration_track': max_duration_track,
            'max_rating_track': max_rating_track
        }

    def print_statistics(self) -> None:
        stats = self.get_statistics()
        print("\n" + "=" * 60)
        print(" ОТЧЕТ О ПЛЕЙЛИСТЕ".center(60))
        print("=" * 60)
        if stats['total_tracks'] == 0:
            print("Плейлист пуст")
            return
        print(f"\n Общая информация:")
        print(f"   * Всего треков: {stats['total_tracks']}")
        print(f"   * Общая длительность: {stats['total_duration_formatted']}")
        print(f"   * Средний рейтинг: {stats['average_rating']}")
        print(f"\n Жанровое разнообразие:")
        for genre, count in stats['genres'].items():
            print(f"   * {genre}: {count} трек(ов)")
        print(f"\n Рекорды:")
        if stats['max_duration_track']:
            minutes = stats['max_duration_track'].duration // 60
            seconds = stats['max_duration_track'].duration % 60
            print(f"   * Самый длинный трек: {stats['max_duration_track'].name} ({minutes}:{seconds:02d})")
        if stats['max_rating_track']:
            print(f"   * Трек с max рейтингом: {stats['max_rating_track'].name} ({stats['max_rating_track'].rating})")
        print("=" * 60)

    def play(self) -> None:
        if len(self.tracks) == 0:
            print("[!] Плейлист пуст. Добавьте треки.")
            return
        print("\n" + "=" * 60)
        print(" ВОСПРОИЗВЕДЕНИЕ ПЛЕЙЛИСТА ".center(60))
        print("=" * 60)
        print(f"Режим повтора: {self.get_repeat_mode_display()}")
        print("=" * 60)
        if self.repeat_mode == "one":
            self._play_one_repeat()
        else:
            self._play_normal()

    def _play_normal(self) -> None:
        if self.repeat_mode == "all":
            continue_playing = True
            while continue_playing:
                for i, track in enumerate(self.tracks, 1):
                    print(f"\n {i}. {track}")
                    if i < len(self.tracks):
                        input("   Нажмите Enter для следующего трека...")
                print("\n Плейлист закончился. Начинаем сначала...")
                continue_playing = self._ask_continue()
        else:
            tracks_list = self.tracks.to_list()
            for i, track in enumerate(tracks_list, 1):
                print(f"\n {i}. {track}")
                if i < len(tracks_list):
                    input("   Нажмите Enter для следующего трека...")
            print("\n Воспроизведение завершено")

    def _play_one_repeat(self) -> None:
        tracks_list = self.tracks.to_list()
        i = 0
        while i < len(tracks_list):
            track = tracks_list[i]
            print(f"\n {i+1}. {track}")
            print("   Команды: n - следующий, r - повторить, p - предыдущий, q - выйти")
            while True:
                choice = input("   Ваш выбор: ").lower()
                if choice == 'n':
                    i += 1
                    break
                elif choice == 'r':
                    print(f"\n ПОВТОР: {track}")
                elif choice == 'p':
                    if i > 0:
                        i -= 1
                        break
                    else:
                        print("    Это первый трек, предыдущего нет")
                elif choice == 'q':
                    return
                else:
                    print("   Неверная команда. Используйте: n, r, p, q")

    def _ask_continue(self) -> bool:
        choice = input("\nПродолжить воспроизведение? (y/n): ").lower()
        return choice == 'y'
```

#Интерактивная фильтрация треков по рейтингу
    def get_tracks_by_rating_interactive(self):
        self.print_header("ТРЕКИ ПО РЕЙТИНГУ")
        if len(self.playlist.tracks) == 0:
            print("\n[!] Плейлист пуст!")
            self.wait_for_enter()
            return
        try:
            min_rating = float(input("\nВведите минимальный рейтинг (0-5): "))
            if 0 <= min_rating <= 5:
                self.playlist.get_tracks_by_rating(min_rating)
            else:
                print("[!] Рейтинг должен быть от 0 до 5!")
        except ValueError:
            print("[!] Введите число!")
        self.wait_for_enter()

    def run(self):
